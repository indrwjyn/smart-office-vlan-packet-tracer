! CORE-SW - Cisco 3560 Multilayer Switch (Packet Tracer 8.2.0.0162)
! Disusun ulang dari CLI dan running-config proyek.
hostname CORE-SW
!
vlan 10
 name MANAGEMENT
vlan 20
 name STAFF
vlan 30
 name FINANCE
vlan 40
 name HRD
vlan 50
 name GUEST
!
ip routing
!
! DHCP (VLAN 10 full static, jadi tidak ada pool)
ip dhcp excluded-address 192.168.20.1 192.168.20.10
ip dhcp excluded-address 192.168.30.1 192.168.30.10
ip dhcp excluded-address 192.168.40.1 192.168.40.10
ip dhcp excluded-address 192.168.50.1 192.168.50.10
ip dhcp pool STAFF_POOL
 network 192.168.20.0 255.255.255.0
 default-router 192.168.20.1
 dns-server 203.0.113.2
ip dhcp pool FINANCE_POOL
 network 192.168.30.0 255.255.255.0
 default-router 192.168.30.1
 dns-server 203.0.113.2
ip dhcp pool HRD_POOL
 network 192.168.40.0 255.255.255.0
 default-router 192.168.40.1
 dns-server 203.0.113.2
ip dhcp pool GUEST_POOL
 network 192.168.50.0 255.255.255.0
 default-router 192.168.50.1
 dns-server 203.0.113.2
!
! Link ke Internet Server (routed port)
interface FastEthernet0/1
 no switchport
 ip address 203.0.113.1 255.255.255.252
!
! Trunk ke access switch
interface range FastEthernet0/2 - 6
 switchport trunk encapsulation dot1q
 switchport mode trunk
!
! Gateway tiap VLAN (SVI)
interface Vlan10
 ip address 192.168.10.1 255.255.255.0
interface Vlan20
 ip address 192.168.20.1 255.255.255.0
interface Vlan30
 ip address 192.168.30.1 255.255.255.0
 ip access-group ACL-FINANCE-OUT out
interface Vlan40
 ip address 192.168.40.1 255.255.255.0
interface Vlan50
 ip address 192.168.50.1 255.255.255.0
 ip access-group 100 in
!
ip route 0.0.0.0 0.0.0.0 203.0.113.2
!
! ACL Guest: hanya boleh ke internet
ip access-list extended 100
 5 permit udp any eq bootpc any eq bootps
 10 deny ip 192.168.50.0 0.0.0.255 192.168.10.0 0.0.0.255
 20 deny ip 192.168.50.0 0.0.0.255 192.168.20.0 0.0.0.255
 30 deny ip 192.168.50.0 0.0.0.255 192.168.30.0 0.0.0.255
 40 deny ip 192.168.50.0 0.0.0.255 192.168.40.0 0.0.0.255
 50 permit ip 192.168.50.0 0.0.0.255 any
!
! ACL Finance: hanya VLAN 10 dan VLAN 30 yang boleh masuk
ip access-list extended ACL-FINANCE-OUT
 permit ip 192.168.10.0 0.0.0.255 192.168.30.0 0.0.0.255
 permit ip 192.168.30.0 0.0.0.255 192.168.30.0 0.0.0.255
 permit icmp any 192.168.30.0 0.0.0.255 echo-reply
 permit tcp any 192.168.30.0 0.0.0.255 established
 deny ip any any
!
end
